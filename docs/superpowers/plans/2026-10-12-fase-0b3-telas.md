# Fase 0b.3: Telas do protótipo — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir as telas de login, contas, categorias, extrato de lançamentos e dashboard sobre a camada base, navegáveis com a API mock, e fechar a fase com o checkpoint de validação do design.

**Architecture:** Uma pasta por feature em `src/app/features/<nome>` (rota lazy + escopo do Transloco), com `data/` (modelo, API, lookup), `list/` (página) e `form/` (formulário em diálogo). As telas de cadastro são majoritariamente declaração (colunas, filtros, ações) sobre `injectCrudList`, `<app-data-table>`, `<app-filter-bar>` e `CrudActions`. Features só importam umas das outras pelo `index.ts`.

**Tech Stack:** Angular 21, PrimeNG 21 (`p-treetable`, `p-tabs`, `p-select`, `p-datepicker`, `p-inputnumber`, `p-password`, `p-menu`), Transloco (escopos por rota), ngx-echarts 21 + echarts (import modular), Vitest.

**Spec:** `docs/superpowers/specs/2026-10-12-fundacao-e-prototipo-design.md` (seções 5, 6 e 8)

**Sequência dos planos:** 4º de 4. Depende de `2026-10-12-fase-0a-fundacao.md`, `2026-10-12-fase-0b1-camada-base.md` e `2026-10-12-fase-0b2-api-mock.md`.

## Global Constraints

- Todas as regras de `2026-10-12-fase-0a-fundacao.md` e `2026-10-12-fase-0b1-camada-base.md` continuam valendo.
- Uma feature importa outra **só** por `@features/<nome>` (o `index.ts`); o lint barra `@features/<nome>/<interno>`.
- Textos das features ficam no escopo do Transloco (`public/i18n/<feature>/pt-BR.json`), com chaves prefixadas pelo escopo (`accounts.fields.name`). Enums usam as chaves globais `enums.*`.
- Nenhum cálculo financeiro no app: saldos e totais vêm da API (no protótipo, do mock).
- Para ver as telas: `npm run start:mock` e abrir `http://localhost:4200` (login com qualquer e-mail/senha).

import type { ApiResponse, ApiSchema } from '@api/api-types';
import type { LookupOption } from '@shared/crud/filter';

export type Account = ApiResponse<'AccountResponse', 'institution'>;
export type AccountSummary = ApiResponse<'AccountSummary'>;
export type AccountBalance = ApiResponse<'AccountBalanceResponse'>;
export type CreateAccountRequest = ApiSchema<'CreateAccountRequest'>;
export type UpdateAccountRequest = ApiSchema<'UpdateAccountRequest'>;
export type AccountType = Account['type'];

/** Query params de GET /accounts e /accounts/options. */
export interface AccountFilter {
  name: string;
  type: AccountType;
  archived: boolean;
}

/** O satisfies quebra a compilação se o backend adicionar ou remover um tipo. */
export const ACCOUNT_TYPES = [
  'CHECKING',
  'SAVINGS',
  'WALLET',
  'INVESTMENT',
  'OTHER',
] as const satisfies readonly AccountType[];

export function toAccountOptions(accounts: readonly AccountSummary[]): LookupOption[] {
  return accounts.map((account) => ({ value: account.id, label: account.name }));
}

import { Injectable } from '@angular/core';
import { CrudApi } from '@shared/crud/crud-api';
import { toHttpParams } from '@shared/data/http-params';
import type { PageRequest, PageResponse } from '@shared/data/page';
import type { Observable } from 'rxjs';

import type {
  Account,
  AccountBalance,
  AccountFilter,
  AccountSummary,
  CreateAccountRequest,
  UpdateAccountRequest,
} from './account.model';

@Injectable({ providedIn: 'root' })
export class AccountApi extends CrudApi<Account, CreateAccountRequest, UpdateAccountRequest, AccountFilter> {
  protected readonly path = '/accounts';

  options(filter: Partial<AccountFilter>, page: PageRequest): Observable<PageResponse<AccountSummary>> {
    return this.http.get<PageResponse<AccountSummary>>(`${this.url}/options`, { params: toHttpParams(filter, page) });
  }

  balance(id: string): Observable<AccountBalance> {
    return this.http.get<AccountBalance>(`${this.url}/${id}/balance`);
  }
}

import { InputNumber } from 'primeng/inputnumber';
import { InputText } from 'primeng/inputtext';
import { Select } from 'primeng/select';

import { AccountApi } from '../data/account.api';
import type { AccountType } from '../data/account.model';
import { ACCOUNT_TYPES } from '../data/account.model';

@Component({
  selector: 'app-account-form',
  imports: [ReactiveFormsModule, FieldComponent, FormFooterComponent, InputNumber, InputText, Select],
  changeDetection: ChangeDetectionStrategy.OnPush,
  templateUrl: './account-form.component.html',
})
export class AccountFormComponent {
  private readonly api = inject(AccountApi);
  private readonly transloco = inject(TranslocoService);
  protected readonly ref = inject(DynamicDialogRef);
  private readonly id = (inject(DynamicDialogConfig).data as { id?: string } | undefined)?.id;

  protected readonly typeOptions = ACCOUNT_TYPES.map((value) => ({
    value,
    label: this.transloco.translate(`enums.accountType.${value}`),
  }));

  protected readonly form = inject(NonNullableFormBuilder).group({
    name: ['', [Validators.required, Validators.maxLength(100)]],
    type: ['CHECKING' as AccountType, Validators.required],
    institution: ['', Validators.maxLength(100)],
    initialBalance: [0, Validators.required],
  });

  protected readonly submit = injectFormSubmit(
    this.form,
    (value, context) => {
      const body = { ...value, institution: value.institution.trim() || undefined };
      return this.id ? this.api.update(this.id, body, context) : this.api.create(body, context);
    },
    () => this.ref.close(true),
  );

  constructor() {
    if (this.id) {
      this.api
        .get(this.id)
        .subscribe((account) => this.form.patchValue({ ...account, institution: account.institution ?? '' }));
    }
  }
}

import type { LookupOption } from '@shared/crud/filter';
import type { TreeNode } from 'primeng/api';

import type { Category, CategorySummary, CategoryType } from './category.model';

interface CategoryLike {
  id: string;
  name: string;
  parentId: string | null;
}

const byName = (a: CategoryLike, b: CategoryLike) => a.name.localeCompare(b.name, 'pt-BR');

/** Raízes ordenadas por nome, cada uma seguida das filhas. Filha cujo pai não veio na lista vira raiz. */
function inTreeOrder<T extends CategoryLike>(items: readonly T[]): { item: T; depth: number }[] {
  const ids = new Set(items.map((c) => c.id));
  const roots = items.filter((c) => !c.parentId || !ids.has(c.parentId)).sort(byName);
  return roots.flatMap((root) => [
    { item: root, depth: 0 },
    ...items
      .filter((c) => c.parentId === root.id)
      .sort(byName)
      .map((child) => ({ item: child, depth: 1 })),
  ]);
}

/** Árvore para o p-treetable da tela de categorias (2 níveis, expandida). */
export function buildCategoryTree(items: readonly Category[]): TreeNode<Category>[] {
  const nodes: TreeNode<Category>[] = [];
  for (const { item, depth } of inTreeOrder(items)) {
    const node: TreeNode<Category> = { key: item.id, data: item, children: [], expanded: true };
    if (depth === 0) nodes.push(node);
    else nodes[nodes.length - 1].children?.push(node);
  }
  return nodes;
}

/** Opções de seletor em ordem de árvore, com recuo nas subcategorias e a cor da categoria. */
export function toCategoryOptions(items: readonly CategorySummary[], type?: CategoryType): LookupOption[] {
  const ofType = type ? items.filter((c) => c.type === type) : items;
  return inTreeOrder(ofType).map(({ item, depth }) => ({
    value: item.id,
    label: item.name,
    depth,
    color: item.color,
  }));
}

import type { ApiResponse, ApiSchema } from '@api/api-types';
import type { AccountSummary } from '@features/accounts';
import type { CategorySummary } from '@features/categories';
import type { LocalDate } from '@shared/data/dates';
import { toLocalDate } from '@shared/data/dates';

type TransactionBase = ApiResponse<
  'TransactionResponse',
  'creditCardInvoiceId' | 'installmentGroupId' | 'installmentNumber' | 'installmentTotal' | 'notes'
>;

/** Lançamento como a API devolve (conta, destino e categoria vêm resumidos). */
export type Transaction = Omit<TransactionBase, 'account' | 'destinationAccount' | 'category'> & {
  account: AccountSummary;
  destinationAccount: AccountSummary | null;
  category: CategorySummary | null;
};
export type CreateTransactionRequest = ApiSchema<'CreateTransactionRequest'>;
export type UpdateTransactionRequest = ApiSchema<'UpdateTransactionRequest'>;
export type CreateTransferRequest = ApiSchema<'CreateTransferRequest'>;
export type TransactionType = Transaction['type'];
export type TransactionStatus = Transaction['status'];

/** Query params de GET /transactions. */
export interface TransactionFilter {
  from: LocalDate;
  to: LocalDate;
  accountId: string;
  categoryId: string;
  creditCardInvoiceId: string;
  type: TransactionType;
  status: TransactionStatus;
  description: string;
}

export const TRANSACTION_TYPES = ['INCOME', 'EXPENSE', 'TRANSFER'] as const satisfies readonly TransactionType[];
export const TRANSACTION_STATUSES = ['PENDING', 'CLEARED'] as const satisfies readonly TransactionStatus[];

export type ReadOnlyReason = 'card' | 'system';

/**
 * Lançamentos de cartão e de categoria de sistema não são editados pelo extrato. O lançamento não diz se a categoria é
 * de sistema; ela é a que não aparece em /categories/options (knownCategoryIds). Enquanto as categorias não carregam
 * (null), só o cartão é considerado.
 */
export function readOnlyReason(t: Transaction, knownCategoryIds: ReadonlySet<string> | null): ReadOnlyReason | null {
  if (t.creditCardInvoiceId) return 'card';
  if (knownCategoryIds && t.category && !knownCategoryIds.has(t.category.id)) return 'system';
  return null;
}

/** Corpo do PUT a partir do lançamento atual (ex.: efetivar = mudar só o status). */
export function toUpdateRequest(
  t: Transaction,
  changes: Partial<UpdateTransactionRequest> = {},
): UpdateTransactionRequest {
  return {
    description: t.description,
    amount: t.amount,
    date: t.date,
    status: t.status,
    accountId: t.account.id,
    destinationAccountId: t.destinationAccount?.id,
    categoryId: t.category?.id,
    notes: t.notes ?? undefined,
    ...changes,
  };
}

/** Mesma regra do backend quando o status não é informado: data futura é PENDING. */
export function statusForDate(date: Date, today = new Date()): TransactionStatus {
  return toLocalDate(date) > toLocalDate(today) ? 'PENDING' : 'CLEARED';
}

        status: value.status,
        notes: value.notes.trim() || undefined,

            { ...common, accountId: value.fromAccountId, destinationAccountId: value.toAccountId },

            context,
          );

    () => this.ref.close(true),

        toAccountId: t.destinationAccount?.id ?? '',
        description: t.description,
        amount: t.amount,
        date: fromLocalDate(t.date),
        status: t.status,
        notes: t.notes ?? '',

      if (date && !this.form.controls.status.dirty) this.form.controls.status.setValue(statusForDate(date));

          value: c.amount,
          itemStyle: c.color ? { color: c.color } : undefined,
        })),
      },
    ],
  }));

  constructor() {
    effect((onCleanup) => {
      const month = toYearMonth(this.month());
      this.loading.set(true);
      const subscription = forkJoin({
        summary: this.api.summary(month),
        expenses: this.api.expensesByCategory(month),
      })
        .pipe(finalize(() => this.loading.set(false)))
        .subscribe(({ summary, expenses }) => {
          this.summary.set(summary);
          this.expenses.set(expenses);
        });
      onCleanup(() => subscription.unsubscribe());
    });
  }

  protected shiftMonth(delta: number): void {
    this.month.update((date) => new Date(date.getFullYear(), date.getMonth() + delta, 1));
  }

  protected selectMonth(date: Date | null): void {
    if (date) this.month.set(firstOfMonth(date));
  }
}
